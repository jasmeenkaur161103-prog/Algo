<script>

function calculateSMA(prices, period, index) {

    if (index < period - 1) {
        return null;
    }

    let sum = 0;

    for (let i = index - period + 1; i <= index; i++) {
        sum += prices[i];
    }

    return sum / period;
}


function runStrategy() {

    // Get prices
    let priceInput = document.getElementById("prices").value;

    let prices = priceInput
        .split(",")
        .map(Number)
        .filter(x => !isNaN(x));

    let shortPeriod =
        parseInt(document.getElementById("shortPeriod").value);

    let longPeriod =
        parseInt(document.getElementById("longPeriod").value);


    if (prices.length < longPeriod) {

        alert(
            "Please enter at least " +
            longPeriod +
            " prices."
        );

        return;
    }


    if (shortPeriod >= longPeriod) {

        alert(
            "Short SMA period must be smaller than Long SMA period."
        );

        return;
    }


    let results = document.getElementById("results");

    results.innerHTML = "";


    let previousShort = null;
    let previousLong = null;

    let latestSignal = "HOLD";


    for (let i = 0; i < prices.length; i++) {

        let shortSMA =
            calculateSMA(prices, shortPeriod, i);

        let longSMA =
            calculateSMA(prices, longPeriod, i);

        let signal = "HOLD";


        // Detect crossover
        if (
            shortSMA !== null &&
            longSMA !== null &&
            previousShort !== null &&
            previousLong !== null
        ) {

            // Bullish crossover
            if (
                previousShort <= previousLong &&
                shortSMA > longSMA
            ) {

                signal = "BUY";
                latestSignal = "BUY";
            }


            // Bearish crossover
            else if (
                previousShort >= previousLong &&
                shortSMA < longSMA
            ) {

                signal = "SELL";
                latestSignal = "SELL";
            }
        }


        let row = document.createElement("tr");


        row.innerHTML = `
            <td>${i + 1}</td>

            <td>${prices[i].toFixed(2)}</td>

            <td>
                ${
                    shortSMA !== null
                    ? shortSMA.toFixed(2)
                    : "-"
                }
            </td>

            <td>
                ${
                    longSMA !== null
                    ? longSMA.toFixed(2)
                    : "-"
                }
            </td>

            <td class="${
                signal === "BUY"
                ? "buy-text"
                : signal === "SELL"
                ? "sell-text"
                : "hold-text"
            }">
                ${signal}
            </td>
        `;


        results.appendChild(row);


        if (shortSMA !== null) {
            previousShort = shortSMA;
        }

        if (longSMA !== null) {
            previousLong = longSMA;
        }
    }


    // Display latest signal

    let signalBox =
        document.getElementById("currentSignal");


    signalBox.className = "signal";


    if (latestSignal === "BUY") {

        signalBox.classList.add("buy");

        signalBox.innerHTML =
            "🟢 BUY SIGNAL";

    }

    else if (latestSignal === "SELL") {

        signalBox.classList.add("sell");

        signalBox.innerHTML =
            "🔴 SELL SIGNAL";

    }

    else {

        signalBox.classList.add("hold");

        signalBox.innerHTML =
            "🟡 HOLD";
    }
}
